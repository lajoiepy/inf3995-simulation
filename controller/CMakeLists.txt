cmake_minimum_required(VERSION 3.16)
project(inf3995_go2)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
find_package(gz-sim8 REQUIRED)
find_package(Eigen3 REQUIRED)
file(GLOB_RECURSE GUIDE_SOURCES vendor/guide/src/*.cpp vendor/guide/src/*.cc)
add_library(inf3995-go2-controller SHARED src/Go2Controller.cc ${GUIDE_SOURCES})
target_include_directories(inf3995-go2-controller PRIVATE vendor/guide/include)
target_compile_definitions(inf3995-go2-controller PRIVATE ROBOT_TYPE_Go1 COMPILE_WITH_SIMULATION)
target_link_libraries(inf3995-go2-controller PRIVATE gz-sim8::gz-sim8 Eigen3::Eigen)
install(TARGETS inf3995-go2-controller LIBRARY DESTINATION lib)
